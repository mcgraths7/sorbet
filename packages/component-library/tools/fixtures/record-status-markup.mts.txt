/** Props as data, made React props: "noop" for onDismiss is a function that does nothing; a child { element, props } is one host element. */
function live(props: Props): Props {
  const made: Props = {};
  for (const [key, value] of Object.entries(props)) {
    if (key === "onDismiss" && value === "noop") {
      made[key] = () => {};
    } else if (key === "children" && typeof value === "object" && value !== null && "element" in value) {
      const child = value as { element: string; props: Props };
      made[key] = React.createElement(child.element, child.props);
    } else {
      made[key] = value;
    }
  }
  return made;
}

const cases = CASES.map(({ id, component, props }) => {
  const Component = barrel[component];
  if (typeof Component !== "function") {
    throw new Error(`${id}: 9b83e50's barrel exports no ${component}`);
  }
  return { id, component, props, html: renderToStaticMarkup(React.createElement(Component, live(props))) };
});
if (cases.length !== 39) {
  throw new Error(`${cases.length} cases, not L187's 39`);
}
